"use client";

import { useEffect, useRef, useState } from "react";
import { gsap } from "gsap";
import Link from "next/link";
import { hero } from "@/data/content";

// Three photographs travel around an ellipse. Paint, photographs and type occupy
// separate depth planes; the image files themselves are never distorted or edited.
export default function Hero() {
  const root = useRef<HTMLElement>(null);
  const advance = useRef<(direction: number) => void>(() => {});
  const pointer = useRef({ x: 0, y: 0 });
  const [active, setActive] = useState(0);
  const [paused, setPaused] = useState(false);
  const pauseRef = useRef(false);
  const focusRef = useRef(false);
  const touchStart = useRef<number | null>(null);

  useEffect(() => { pauseRef.current = paused; }, [paused]);

  useEffect(() => {
    const el = root.current!;
    const cards = [...el.querySelectorAll<HTMLElement>("[data-orbit-card]")];
    const paints = [...el.querySelectorAll<HTMLElement>("[data-orbit-paint]")];
    const media = window.matchMedia("(prefers-reduced-motion: reduce)");
    const state = { angle: 0, x: 0, y: 0 };
    let current = 0;
    let visible = true;
    let busy = false;
    let nextAt = performance.now() + 7500;
    let tween: gsap.core.Tween | undefined;

    const render = () => {
      const mobile = el.clientWidth < 768;
      const radius = el.clientWidth * (mobile ? .77 : .60);
      state.x += (pointer.current.x - state.x) * .065;
      state.y += (pointer.current.y - state.y) * .065;
      cards.forEach((card, i) => {
        const angle = (i * 120 + state.angle) * Math.PI / 180;
        const depth = (Math.cos(angle) + 1) / 2;
        const scale = .33 + depth * .67;
        const x = Math.sin(angle) * radius - state.x * (10 + depth * 16);
        const y = (1 - depth) * -el.clientHeight * .075 + state.y * (5 + depth * 8);
        card.style.transform = `translate(-50%, -50%) translate3d(${x}px, ${y}px, 0) scale(${scale}) perspective(1000px) rotateY(${-state.x * depth * 7}deg) rotateZ(${[-4, 3, 4][i]}deg)`;
        card.style.zIndex = String(Math.round(depth * 10) + 2);
        card.style.filter = `blur(${(1 - depth) * 5}px)`;
        card.style.opacity = String(.24 + depth * .76);
        paints[i].style.transform = `translate(-50%, -50%) translate3d(${Math.sin(angle) * radius * .78 - state.x * 7}px, ${state.y * 5}px, 0) scale(${.60 + depth * .5}) rotate(${i * 28 + state.angle * .12}deg)`;
        paints[i].style.opacity = String(.12 + depth * .30);
      });
    };

    const rotate = (direction: number) => {
      if (busy) return;
      const target = state.angle - direction * 120;
      current = (current + direction + 3) % 3;
      setActive(current);
      nextAt = performance.now() + 6000;
      tween?.kill();
      if (media.matches) { state.angle = target; render(); return; }
      busy = true;
      tween = gsap.to(state, { angle: target, duration: 1.4, ease: "expo.inOut", onComplete: () => { busy = false; } });
    };
    advance.current = rotate;
    const tick = () => {
      if (document.hidden || !visible) return;
      if (!media.matches) render();
      if (!media.matches && !pauseRef.current && !focusRef.current && !busy && performance.now() > nextAt && document.documentElement.classList.contains("is-loaded")) rotate(1);
    };
    const observer = new IntersectionObserver(([entry]) => {
      visible = entry.isIntersecting;
      nextAt = performance.now() + 6000;
    }, { threshold: .15 });
    observer.observe(el);
    const onVisibility = () => { nextAt = performance.now() + 6000; };
    const resize = new ResizeObserver(render);
    resize.observe(el);
    const onMotionChange = () => {
      tween?.kill(); busy = false; state.angle = -current * 120;
      pointer.current = { x: 0, y: 0 }; state.x = 0; state.y = 0;
      render();
    };
    media.addEventListener("change", onMotionChange);
    document.addEventListener("visibilitychange", onVisibility);
    gsap.ticker.add(tick);
    render();
    return () => {
      advance.current = () => {};
      tween?.kill(); gsap.ticker.remove(tick); observer.disconnect(); resize.disconnect();
      media.removeEventListener("change", onMotionChange);
      document.removeEventListener("visibilitychange", onVisibility);
    };
  }, []);

  return (
    <section ref={root} id="hero" className="orbit-hero" aria-label="日常物件，三件展品" aria-roledescription="輪播"
      onPointerMove={(e) => {
        if (e.pointerType !== "mouse") return;
        const r = e.currentTarget.getBoundingClientRect();
        pointer.current = { x: (e.clientX - r.left) / r.width * 2 - 1, y: (e.clientY - r.top) / r.height * 2 - 1 };
      }}
      onPointerLeave={() => { pointer.current = { x: 0, y: 0 }; }}
      onFocusCapture={() => { focusRef.current = true; }}
      onBlurCapture={(e) => { if (!e.currentTarget.contains(e.relatedTarget)) focusRef.current = false; }}
      onTouchStart={(e) => { touchStart.current = e.touches[0].clientX; }}
      onTouchEnd={(e) => {
        if (touchStart.current !== null) {
          const delta = e.changedTouches[0].clientX - touchStart.current;
          if (Math.abs(delta) > 45) advance.current(delta < 0 ? 1 : -1);
        }
        touchStart.current = null;
      }}>
      <div className="orbit-stage" data-hero-reveal="">
        <div className="orbit-paints" aria-hidden="true">
          {hero.slides.map((slide, i) => (
            <div className={`orbit-paint orbit-paint--${i}`} data-orbit-paint="" key={slide.id}>
              <svg viewBox="0 0 900 1000" fill="none">
                <defs>
                  <filter id={`drybrush-${i}`} x="-15%" y="-15%" width="130%" height="130%">
                    <feTurbulence type="fractalNoise" baseFrequency=".035 .16" numOctaves="3" seed={12 + i} result="grain" />
                    <feDisplacementMap in="SourceGraphic" in2="grain" scale="22" xChannelSelector="R" yChannelSelector="G" />
                  </filter>
                </defs>
                <g stroke="currentColor" filter={`url(#drybrush-${i})`}>
                  <path d="M72 1005C188 813 330 466 609 192C759 44 781 138 685 353C592 562 344 859 393 714C437 587 626 350 808 262" strokeWidth="62" />
                  <path d="M42 995C160 793 306 442 583 176M93 1020C247 784 407 387 691 124M378 750C436 601 674 324 831 271" strokeWidth="8" />
                  <path d="M120 987C256 723 388 373 646 158M396 785C460 619 676 354 823 293" strokeWidth="4" />
                </g>
              </svg>
            </div>
          ))}
        </div>
        <div className="orbit-photographs">
          {hero.slides.map((slide, i) => (
            <div key={slide.id} data-orbit-card="" data-active={active === i} className={`orbit-card orbit-card--${slide.id}`} aria-hidden={active !== i}>
              <Link href={slide.href} tabIndex={active === i ? 0 : -1} aria-label={`探索${slide.label}`} className="orbit-photo-link">
                <img src={slide.src} alt={slide.alt} width={slide.w} height={slide.h} fetchPriority={i === 0 ? "high" : "auto"} loading="eager" draggable={false} />
              </Link>
            </div>
          ))}
        </div>
      </div>
      <h1 className="orbit-title" data-hero-reveal=""><span>Everyday</span><span>Luxuries.</span></h1>
      <p className="orbit-statement tc" data-hero-reveal=""><span>把日常的物件，</span><span>當作展品。</span></p>
      <div className="orbit-caption" data-hero-reveal="">
        <span className="orbit-kicker">Selected objects</span>
        <Link href={hero.slides[active].href} className="orbit-object-link"><span>{hero.slides[active].en}</span><span aria-hidden="true">↗</span></Link>
        <span className="tc">{hero.slides[active].label}</span>
      </div>
      <div className="orbit-footer">
        <span className="orbit-location">CHARM VILLA · TAIPEI / KYOTO</span>
        <div className="orbit-controls" aria-label="展品輪播控制">
          <button type="button" aria-label="上一張" onClick={() => advance.current(-1)}>←</button>
          <span className="orbit-count" aria-live="off">0{active + 1}<span> / 03</span></span>
          <button type="button" aria-label="下一張" onClick={() => advance.current(1)}>→</button>
          <button type="button" className="orbit-pause" aria-label={paused ? "播放輪播" : "暫停輪播"} aria-pressed={paused} onClick={() => setPaused(v => !v)}>{paused ? "▷" : "Ⅱ"}</button>
        </div>
        <a className="orbit-scroll" href="#manifesto">Scroll <span aria-hidden="true">↓</span></a>
      </div>
    </section>
  );
}
